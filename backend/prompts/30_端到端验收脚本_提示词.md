## 1. 任务目标

写一个**一键运行的 Python 脚本**，加载 B-07 测试画像 + B-06 知识库，跑通 D-03 协同流程，**自动计算 3 项硬指标并生成报告**。

**这是夺奖最关键的"证据"，必须在 9-5 截止前能跑通。**

---

## 2. 必达指标

| 指标 | 目标 |
| --- | --- |
| 幻觉率（hallucination rate） | **< 5%** |
| 画像-难度适配准确率 | **≥ 85%** |
| 核心知识点覆盖率 | **≥ 90%** |
| 报告输出 | `docs/quality_reports/quality_report_YYYYMMDD.md` |
| 失败处理 | 任一指标不达标 → 抛 `QualityError` 并退出码 1 |

## 3. 脚本结构

```python
# backend/a_用户与聊天/quality_check.py
"""
端到端验收脚本
- 加载 B-07 的 3 组测试画像
- 对每组画像跑 D-03 协同流程
- 计算 3 项硬指标
- 输出报告
- 任一指标不达标抛 QualityError
"""
import json
from datetime import datetime
from pathlib import Path
from backend.d_AI集成.orchestrator.orchestrate import orchestrate
from backend.公共.metrics import (
    calc_hallucination_rate,
    calc_match_accuracy,
    calc_coverage,
)

class QualityError(Exception):
    pass

def quality_check(
    profiles_path: str = "backend/b_学情数据/test_profiles",
    kb_path: str = "backend/b_学情数据/knowledge_base",
    output_dir: str = "docs/quality_reports"
):
    # 1. 加载 3 组测试画像
    profiles = []
    for p in Path(profiles_path).glob("*.json"):
        profiles.append(json.loads(p.read_text()))
    
    assert len(profiles) == 3, f"期望 3 组画像，实际 {len(profiles)}"
    
    # 2. 对每组画像跑 D-03 协同流程
    results = []
    for profile in profiles:
        resource = orchestrate(
            studentId=profile["student_id"],
            resource_type="customized_resource"
        )
        results.append({
            "profile": profile,
            "resource": resource,
        })
    
    # 3. 计算 3 项硬指标
    hallucination = calc_hallucination_rate(results, kb_path)
    accuracy = calc_match_accuracy(results)
    coverage = calc_coverage(results, kb_path)
    
    # 4. 输出报告
    report = f"""# 质量报告 - {datetime.now():%Y%m%d}

## 3 项硬指标
| 指标 | 实际 | 目标 | 状态 |
| --- | --- | --- | --- |
| 幻觉率 | {hallucination:.2%} | < 5% | {'✅' if hallucination < 0.05 else '❌'} |
| 画像-难度适配准确率 | {accuracy:.2%} | ≥ 85% | {'✅' if accuracy >= 0.85 else '❌'} |
| 核心知识点覆盖率 | {coverage:.2%} | ≥ 90% | {'✅' if coverage >= 0.90 else '❌'} |

## 详细数据
（共 {len(results)} 组画像，每组已跑通完整协同流程）
"""
    
    output_path = Path(output_dir) / f"quality_report_{datetime.now():%Y%m%d_%H%M%S}.md"
    output_path.parent.mkdir(parents=True, exist_ok=True)
    output_path.write_text(report, encoding="utf-8")
    print(f"报告已生成：{output_path}")
    
    # 5. 任一指标不达标抛 QualityError
    if hallucination >= 0.05 or accuracy < 0.85 or coverage < 0.90:
        raise QualityError(
            f"3 项硬指标未达标：幻觉率={hallucination:.2%}, "
            f"准确率={accuracy:.2%}, 覆盖率={coverage:.2%}"
        )

if __name__ == "__main__":
    quality_check()
```

## 4. 3 项指标的计算函数（S-01 公共任务）

```python
# backend/公共/metrics.py
from typing import List, Dict

def calc_hallucination_rate(results: List[Dict], kb_path: str) -> float:
    """
    幻觉率 = 资源中切片外内容占比
    = sum(切片外内容字数) / sum(资源总字数)
    """
    total_words = 0
    hallucinated_words = 0
    for r in results:
        content = json.dumps(r["resource"].content, ensure_ascii=False)
        cited = set(r["resource"].cited_chunks)
        # 简化实现：每个 chunk 内容已知，超出 cited 的视为幻觉
        # 实际可用 LLM 二次审核或 NLI 模型
        ...
    return hallucinated_words / total_words if total_words else 0

def calc_match_accuracy(results: List[Dict]) -> float:
    """
    画像-难度适配准确率
    = 推荐难度与 expected 难度一致的画像占比
    """
    matched = 0
    for r in results:
        if r["resource"].recommendedDifficulty == r["profile"]["expected"]["recommendedDifficulty"]:
            matched += 1
    return matched / len(results)

def calc_coverage(results: List[Dict], kb_path: str) -> float:
    """
    核心知识点覆盖率
    = 资源覆盖的 kp 数 / kp_taxonomy 总数
    """
    import json
    kp_taxonomy = json.loads(Path(kb_path, "kp_taxonomy.json").read_text())
    total_kps = len(kp_taxonomy["kps"])
    
    covered = set()
    for r in results:
        covered.update(r["resource"].kp_coverage)
    return len(covered) / total_kps
```
